async function heroPhoto(): Promise<string> {
  const [found] = await db
    .select({ p: mediaAssets.storagePath })
    .from(mediaAssets)
    .where(and(eq(mediaAssets.sellerId, SELLER_ID), eq(mediaAssets.filename, "romana-heart-gloss.jpeg")))
    .limit(1);
  if (found) return media(found.p);
  if (MODE === "dry") return media("PENDING");
  const path = `${SELLER_ID}/${STORE_ID}/${crypto.randomUUID()}.jpg`;
  const { error } = await getServiceClient()
    .storage.from("store-media")
    .upload(path, readFileSync("C:/Users/elsha/OneDrive/Desktop/tallaby/romana store/Lip gloss 150 EGP (2).jpeg"), { contentType: "image/jpeg", cacheControl: "31536000", upsert: false });
  if (error) throw new Error("upload " + error.message);
  const rec = await recordMediaAsset(SELLER_ID, STORE_ID, {
    storagePath: path,
    filename: "romana-heart-gloss.jpeg",
    width: 1212,
    height: 1297,
    altText: { en: "Romana heart gloss in berry", ar: "ملمع رومانا بغطاء القلب" },
  });
  if (!rec.success) throw new Error("record " + rec.error);
  return media(path);
}
